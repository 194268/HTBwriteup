import re

def bits_to_string(bit_list):
 res = “”
 # 每 8 bits 提取一次
 for i in range(0, len(bit_list), 8):
 byte_bits = bit_list[i:i+8]
 if len(byte_bits) < 8: break

# 将 bit 列表转为 01 字符串并转为整数
 byte_str = “”.join(map(str, byte_bits))
 res += chr(int(byte_str, 2))
 return res
