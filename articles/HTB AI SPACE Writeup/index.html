import numpy as np
import matplotlib.pyplot as plt

# 1. 读取距离矩阵
D = np.load(“distance_matrix.npy”, allow_pickle=False)
print(“[+] shape:”, D.shape)
print(“[+] dtype:”, D.dtype)

# 2. 基本检查
D = D.astype(np.float64)
print(“[+] diagonal max:”, np.abs(np.diag(D)).max())
print(“[+] symmetric diff:”, np.abs(D — D.T).max())

# 轻微误差修正
D = (D + D.T) / 2
np.fill_diagonal(D, 0)

# 3. Classical MDS：把距离矩阵还原成 2D 坐标
D2 = D ** 2
row_mean = D2.mean(axis=1, keepdims=True)
col_mean = D2.mean(axis=0, keepdims=True)
total_mean = D2.mean()

B = -0.5 * (D2 — row_mean — col_mean + total_mean)

eigvals, eigvecs = np.linalg.eigh(B)
idx = np.argsort(eigvals)[::-1]

eigvals = eigvals[idx]
eigvecs = eigvecs[:, idx]

coords = eigvecs[:, :2] * np.sqrt(np.maximum(eigvals[:2], 0))
x, y = coords[:, 0], coords[:, 1]

# 4. 画四个方向，防止 flag 镜像/倒置
views = [
 ( x, y, “normal”),
 (-x, y, “flip_x”),
 ( x, -y, “flip_y”),
 (-x, -y, “flip_xy”),
]

fig, axes = plt.subplots(2, 2, figsize=(12, 12))

for ax, (px, py, title) in zip(axes.ravel(), views):
 ax.scatter(px, py, s=1)
 ax.set_title(title)
 ax.set_aspect(“equal”)
 ax.axis(“off”)

plt.tight_layout()
plt.savefig(“flag_views.png”, dpi=300)
plt.show()

print(“[+] saved: flag_views.png”)
